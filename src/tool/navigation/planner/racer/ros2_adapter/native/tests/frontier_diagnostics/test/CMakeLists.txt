cmake_minimum_required(VERSION 3.8)
project(frontier_diagnostic_probe)
set(CMAKE_CXX_STANDARD 14)
find_package(ament_cmake REQUIRED)
find_package(active_perception REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED)
add_executable(frontier_probe probe.cpp)
target_include_directories(frontier_probe PRIVATE ${EIGEN3_INCLUDE_DIR} ${PCL_INCLUDE_DIRS})
ament_target_dependencies(frontier_probe active_perception)
target_link_libraries(frontier_probe ${PCL_LIBRARIES})
if(WITH_DIAGNOSTICS)
  target_compile_definitions(frontier_probe PRIVATE WITH_DIAGNOSTICS)
endif()
